{# ============================================================================= VitaNova — shared template macros The four designs differ almost entirely in CSS. These macros give them one common markup vocabulary so a change to, say, how bullets nest applies everywhere at once. ============================================================================= #} {% macro bullets(items) %} {% set filled = items | select | list %} {% if filled %}
    {% for bullet in filled %}
  • {{ bullet }}
    • {% endfor %}
{% endif %} {% endmacro %} {% macro tags(items, separator=' • ') %} {% set filled = items | select | list %} {% if filled %}
{{ filled | join(separator) }}
{% endif %} {% endmacro %} {#- One entry heading: primary text on the left, dates floated right. -#} {% macro entry_row(primary, aside='', primary_class='vn-entry-title', aside_class='vn-entry-dates') %} {% if primary or aside %}
{% if aside %}{{ aside }}{% endif %} {% if primary %}{{ primary }}{% endif %}
{% endif %} {% endmacro %} {#- Experience / education share a shape: title + dates, then org + location. -#} {% macro experience_entry(item) %}
{{ entry_row(item.role, item | date_range) }} {{ entry_row(item.organization, item.location, 'vn-entry-org', 'vn-entry-location') }} {% if item.summary %}
{{ item.summary }}
{% endif %} {{ bullets(item.bullets) }} {{ tags(item.tech) }}
{% endmacro %} {% macro education_entry(item) %}
{{ entry_row(item.degree, item | date_range) }} {{ entry_row(item.institution, item.location, 'vn-entry-org', 'vn-entry-location') }} {{ bullets(item.details) }}
{% endmacro %} {% macro project_entry(item) %}
{{ entry_row(item.name, item.period) }} {{ tags(item.tech) }} {{ bullets(item.bullets) }}
{% endmacro %} {% macro certification_entry(item) %}
{{ entry_row(item.name, item.date) }} {% set detail = [item.issuer, item.note] | select | join(' • ') %} {% if detail %}
{{ detail }}
{% endif %}
{% endmacro %} {% macro language_entry(item) %}
{{ item.name }}
{% if item.level %}
{{ item.level }}
{% endif %}
{% endmacro %} {% macro skill_group(item) %}
{% if item.label %}
{{ item.label }}
{% endif %} {{ tags(item.keywords, ', ') }}
{% endmacro %} {% macro custom_entry(item) %}
{{ entry_row(item.title, item.meta) }} {% if item.subtitle %}
{{ item.subtitle }}
{% endif %} {{ bullets(item.bullets) }}
{% endmacro %} {#- Dispatches a section's items to the right entry macro. Templates call this so none of them has to know about every section type. -#} {% macro section_body(section) %} {% if section.type == 'summary' %}
